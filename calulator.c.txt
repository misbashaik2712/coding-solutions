#include <stdio.h>
int main()
{
char operator;
double num1, num2, result;

printf("enter an operation (+,-,*,/)");
scanf("%c" , &operator);

printf("enter two numbers ");
scanf("%lf %lf", &num1, &num2);

switch (operator)  
{

	case '+':
		result = num1+num2;
				printf("%.2lf + %.2lf = %.2lf\n", num1 , num2, result);
	              	break;
		
	case '-':
		result = num1-num2;
		printf("%.2lf - %.2lf = %.2lf\n", num1 , num2, result);
		break;
		
		case '*':
		result = num1*num2;
		printf("%.2lf * %.2lf = %.2lf\n", num1 , num2, result);
		break;
		
			case '/':
		result = num1/num2;
		printf("%.2lf / %.2lf = %.2lf\n", num1 , num2, result);
		break;
		
		defult:
			printf("error '%c' is an invalid operator.\n", operator);
			break;
		}
		
		return 0 ;
}
